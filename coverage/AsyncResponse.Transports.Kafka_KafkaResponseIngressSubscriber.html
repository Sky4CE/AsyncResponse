 136                Topic,
 137                ConsumerGroup,
 138                SubscriberRole,

 188    private void RunPollLoop(
 189        IKafkaConsumerClient consumer,
 190        KafkaMessageDispatcher dispatcher,
 191        CancellationToken stoppingToken)
 192    {
 193        var paused = false;
 194        var intakeStopLogged = false;
 195        while (!stoppingToken.IsCancellationRequested)
 196        {
 197            // Handlers that outlived their inline budget are settled here, on the poll thread —
 198            // the only thread that touches the consumer — before the next poll: offset stored,
 199            // partition resumed. A settlement that failed for good throws and faults the loop.
 200            dispatcher.SettleCompleted();
 201
 202            KafkaIncomingMessage? message;
 203            if (!dispatcher.CanAcceptMore)
 204            {
 205                // Backpressure — the bounded in-process queue is saturated — or host stop (the
 206                // worker's intake gate closed, and nothing lifts it again): stop fetching from the
 207                // assigned partitions. Keep calling Consume so the broker still sees the consumer
 208                // polling (max.poll.interval.ms) and rebalance callbacks keep firing.
 209                // Re-assert the pause on EVERY such tick, not only on the edge: Pause() snapshots
 210                // the CURRENT assignment, and a partition this member gains meanwhile (from a peer
 211                // leaving the group) arrives unpaused — an edge-triggered pause would leave it
 212                // fetching into the full queue and park the poll thread on the bounded write. (A
 213                // partition the pause already covered keeps it across a revoke and a
 214                // re-assignment: librdkafka holds an application pause on the partition, and the
 215                // adapter lifts it on one that comes back after the resume below.) Pause is a
 216                // local librdkafka call (no broker round trip), so the per-tick re-assert is cheap.
 217                consumer.PauseAssignment();
 218                if (dispatcher.IntakeClosed && !intakeStopLogged)
 219                {
 220                    intakeStopLogged = true;
 221                    SafeLog.Try(() => Logger.LogInformation(
 222                        "Kafka worker subscriber for {Topic} stopped taking new deliveries: the host is stopping. Its as
 223                        Topic));
 224                }
 225                else if (!paused)
 226                {
 227                    SafeLog.Try(() => Logger.LogDebug(
 228                        "Kafka subscriber for {Topic} paused its assignment: the in-process queue is full.",
 229                        Topic));
 230                }
 231
 232                paused = true;
 233
 234                // Once the host is stopping and nothing detached is left to settle, nothing can
 235                // lift the pause any more: a full poll timeout per tick keeps the member alive.
 236                message = consumer.Consume(dispatcher.IntakeClosed && !dispatcher.HasDetachedWork
 237                    ? SubscriberOptions.PollTimeout
 238                    : SubscriberOptions.BackpressurePollDelay);
 239            }
 240            else
 241            {
 242                if (paused)
 243                {
 244                    consumer.ResumeAssignment();
 245                    paused = false;
 246                    SafeLog.Try(() => Logger.LogDebug(
 247                        "Kafka subscriber for {Topic} resumed its assignment: in-process queue capacity freed.",
 248                        Topic));
 249                }
 250
 251                // With detached handlers in flight, poll in short slices so a completion is
 252                // settled within BackpressurePollDelay instead of after a full PollTimeout; the
 253                // slice is what bounds the resume latency of the paused partition.
 254                message = consumer.Consume(dispatcher.HasDetachedWork
 255                    ? SubscriberOptions.BackpressurePollDelay
 256                    : SubscriberOptions.PollTimeout);
 257            }
 258
 259            if (message is null)
 260                continue;
 261
 262            if (dispatcher.IntakeClosed)
 263            {
 264                // Consumed after host stop began — fetched before the pause took hold, or in the
 265                // very poll the gate closed during. Neither started nor settled: its offset is not
 266                // stored (and nothing later of its partition is, since nothing more is taken), so
 267                // the partition's next owner redelivers it. Taken, early ACK committed it at
 268                // enqueue, and a flow wake-up — the very kind this host's own timer hand-overs just
 269                // published for a live replica — ran into the stopping host's hand-back and became
 270                // a dead-letter copy stranding its flow; under ack-after-handler it spent a
 271                // delivery and made the graceful stop wait for it.
 272                SafeLog.Try((Logger, Message: message), static state => state.Logger.LogDebug(
 273                    "Kafka message {Topic}[{Partition}]@{Offset} was consumed after the host began stopping; left unsett
 274                    state.Message.Topic,
 275                    state.Message.Partition,
 276                    state.Message.Offset));
 277                continue;
 278            }
 279
 280            KafkaDelivery delivery;
 281            try
 282            {
 283                delivery = CreateDelivery(message);
 284            }
 285            catch (Exception ex) when (ex is not OperationCanceledException)
 286            {
 287                // A foreign/malformed message can never be handled; dead-letter and commit it so
 288                // its partition advances instead of re-failing on every subscriber restart.
 289                //
 290                // Deliberately every non-cancellation exception, not just InvalidDataException:
 291                // CreateDelivery also runs correlation-id extraction, and anything that escapes
 292                // here faults the poll loop with the offset unstored, so the same message re-throws
 293                // after every supervisor restart and the whole subscriber (all assigned partitions)
 294                // stops advancing — MaxDeliveryAttempts cannot help, because it is keyed on a
 295                // delivery this path never constructed.
 296                //
 297                // Through the dispatcher's partition ordering, not a direct discard: storing this
 298                // message's offset while an earlier message of the same partition is still being
 299                // handled (detached) commits the partition PAST that unfinished message, and a
 300                // crash after the commit skips it for good with no dead-letter copy anywhere.
 301                dispatcher.AcceptUnprocessable(message, ex, stoppingToken);
 302                continue;
 303            }
 304
 305            // Settles inline (queued mode, and ack-after-handler mode within DetachHandlerAfter)
 306            // or detaches the handler and returns; either way the poll thread is back here within
 307            // the validated poll gap.
 308            dispatcher.Accept(delivery, stoppingToken);
 309        }
 310    }
