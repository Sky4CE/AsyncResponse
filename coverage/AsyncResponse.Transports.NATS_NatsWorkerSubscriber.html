 267        List<NatsJobDelivery> batch,
 268        TimeSpan renewalInterval,
 269        CancellationToken stoppingToken)
 270    {
 271        // The batch is dispatched serially, so a slow handler lets the server-side AckWait of the
 272        // later (still unsettled) messages lapse into redelivery. While the batch is in flight, a
 273        // heartbeat signals in-progress for every unsettled message to reset its AckWait window.
 274        // The heartbeat is NOT tied to the stop token: a handler takes no token, so it outlives
 275        // the stop signal, and cancelling its renewal there let AckWait lapse under the live
 276        // handler on every rolling deploy — the job was redelivered to a peer and ran twice. It
 277        // ends only when this loop has let go of every message.
 278        var progress = new BatchProgress();
 279        using var renewalCancellation = new CancellationTokenSource();
 280        var renewalTask = RenewInProgressLoopAsync(batch, progress, renewalInterval, renewalCancellation.Token);
 281        var next = 0;
 282        try
 283        {
 284            for (; next < batch.Count; next++)
 285            {
 286                // Stopping: do not start what has not started. The rest of the batch used to run
 287                // on, handler after handler, past the stop signal. A hand-back from the flow
 288                // engine (inline, or in an early-ACK worker) and host stop closing the worker's
 289                // intake are the same signal, arriving before the token — and from them on
 290                // nothing is settled first (early ACK) either.
 291                if (stoppingToken.IsCancellationRequested || dispatcher.IntakeClosed)
 292                    break;

 294                try
 295                {
 296                    await dispatcher.HandleAsync(batch[next], stoppingToken).ConfigureAwait(false);
 297                }
 298                finally
 299                {
 300                    progress.MarkSettled();
 301                }
 302            }
 303        }
 304        finally
 305        {
 306            // Hand back whatever never started (a stop, or a handler cancelled by it, cut the
 307            // batch short) while the heartbeat still covers it.
 308            for (var i = Math.Max(next, progress.SettledCount); i < batch.Count; i++)
 309                await ReleaseUnstartedAsync(batch[i]).ConfigureAwait(false);
 310
 311            renewalCancellation.Cancel();

 313            {
 314                // Cancellation exits the sweep between messages and aborts the in-flight
 315                // heartbeat (the token reaches the SDK call), so this normally completes at once.
 316                // The bound is the hard backstop for a heartbeat the client cannot abort — a
 317                // write wedged on a dead socket: an unbounded join here held the loop after every
 318                // message in the batch had settled, so no further batch was fetched and a stop
 319                // never completed, with nothing for the supervisor to restart. Past one heartbeat
 320                // interval the loop is abandoned; the server's AckWait settles whatever it left.

 322            }
 323            catch (TimeoutException)

 332                    CancellationToken.None,
 333                    TaskContinuationOptions.OnlyOnFaulted | TaskContinuationOptions.ExecuteSynchronously,
