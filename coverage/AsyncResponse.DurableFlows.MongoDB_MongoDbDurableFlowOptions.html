 323            NotAReplicaSet or InvalidOptions or CommandNotSupported or NotImplemented => true,
 324            MaxTimeMsExpired or ExceededTimeLimit or ReadConcernMajorityNotAvailableYet or LinearizableReadConcernError
 325                or HostUnreachable or HostNotFound or NetworkTimeout or SocketException => false,
 326            // Any other code only when the server says the linearizable level is UNSUPPORTED (a
 327            // service's own answer, e.g. BadValue "unsupported read concern level: linearizable").
 328            // Merely mentioning the level is not enough: MongoDB's own "failed to confirm that read
 329            // was linearizable" is a replica set that cannot confirm right now, and remembering it
 330            // as a refusal would switch the check off for the life of the process.
 331            _ => SaysLinearizableIsUnsupported(exception.ErrorMessage)
 332        };

 362    public async Task<bool> TryCreateAsync(string flowId, FlowState state, TimeSpan ttl, CancellationToken cancellationT
 363    {
 364        DurableFlowStoreShared.ValidateCreate(flowId, state, ttl);
 365        var stateJson = DurableFlowStoreShared.SerializeBounded(flowId, state, _options.MaxStateBytes, "MongoDB");
 366        await EnsureCreatedAsync(cancellationToken).ConfigureAwait(false);
 367
 368        // Two server-side steps instead of one upsert because MongoDB rejects upserts whose query
 369        // uses $expr, and $expr is what lets the expired-check run on the server clock.
 370        //
 371        // Step 1: atomically replace an expired ledger in place. Filter and assignments both
 372        // evaluate on $$NOW, so exactly one competing creator wins and every loser then sees the
 373        // fresh future expiry.
 374        var replaced = await _collection.UpdateOneAsync(
 375            BuildExpiredReplaceFilter(flowId),
 376            BuildStateUpdate(stateJson, state.Revision, ttl, resetLease: true),
 377            options: null,
 378            cancellationToken).ConfigureAwait(false);
 379        if (replaced.ModifiedCount > 0)
 380            return true;
 381
 382        // Step 2: the id was absent (or the expired document was TTL-purged after step 1 looked):
 383        // insert a fresh ledger. A plain insert has no aggregation context, so $$NOW is
 384        // unavailable — instead the server clock is read with one cheap `hello` round-trip and
 385        // stamped client-side. Creation then uses the same authority as every $$NOW comparison
 386        // and refresh below, so app clock skew can never mint a ledger that is born expired or
 387        // outlives its TTL window. A duplicate key means a live ledger owns the id.
 388        var serverNow = await ReadServerNowAsync(cancellationToken).ConfigureAwait(false);
 389        try
 390        {
 391            await _collection.InsertOneAsync(
 392                new MongoFlowStateDocument
 393                {
 394                    FlowId = flowId,
 395                    StateJson = stateJson,
 396                    ExpiresAtUtc = DurableFlowStoreShared.AddSaturating(serverNow, ttl),
 397                    UpdatedAtUtc = serverNow,
 398                    Revision = state.Revision
 399                },
 400                options: null,
 401                cancellationToken).ConfigureAwait(false);
 402            return true;
 403        }
 404        catch (MongoWriteException ex) when (ex.WriteError?.Category == ServerErrorCategory.DuplicateKey)
 405        {
 406            // The document that holds the id may have expired during the `hello` round trip
 407            // between step 1 and this insert. Answering "exists" for it sends the executor to load
 408            // a ledger that reads as absent, and the start job is acknowledged with no run
 409            // created — so replace it once more before conceding the id to a live owner.
 410            var retried = await _collection.UpdateOneAsync(
 411                BuildExpiredReplaceFilter(flowId),
 412                BuildStateUpdate(stateJson, state.Revision, ttl, resetLease: true),
 413                options: null,
 414                cancellationToken).ConfigureAwait(false);
 415            return retried.ModifiedCount > 0;
 416        }
 417    }

 492    private async Task EnsureCreatedAsync(CancellationToken cancellationToken)
 493    {
 494        // No AutoCreateIndexes/UseOwnershipLedger fast-path here: with both disabled there is no
 495        // DDL and no ledger claim, but the TTL-reaper VERIFICATION below must still run once — an
 496        // early-out on the flag pair skipped it for exactly the locked-down deployment
 497        // (operator-provisioned indexes, no ledger writes) it exists to protect, and the
 498        // collection grew without bound with no error and no log line.
 499        if (_created)
 500            return;
 501
 502        await _ensureGate.WaitAsync(cancellationToken).ConfigureAwait(false);
 503        try
 504        {
 505            if (_created)
 506                return;
 507
 508            // Persisted cross-host ownership, independent of AutoCreateIndexes: a flow store
 509            // configured onto the channel's derived counters collection would let this TTL
 510            // index silently delete the ack-sequence counter — see MongoOwnershipLedger.
 511            if (_options.UseOwnershipLedger)
 512            {
 513                await MongoOwnershipLedger.ClaimAsync(
 514                    _database,
 515                    "MongoDB durable-flow store",
 516                    [(_options.CollectionName, nameof(_options.CollectionName))],
 517                    cancellationToken).ConfigureAwait(false);
 518            }
 519
 520            // One index listing serves both startup checks: the collation refusal, and (with
 521            // AutoCreateIndexes off) the TTL verification. With AutoCreateIndexes on, a deployment
 522            // whose credentials may create but not list indexes keeps working — unverified, as
 523            // before the refusal existed.
 524            var indexes = await ListIndexesAsync(tolerateUnauthorized: _options.AutoCreateIndexes, cancellationToken).Co
 525            if (indexes is not null)
 526                ThrowIfFoldingCollation(indexes);
 527
 528            if (!_options.AutoCreateIndexes)
 529            {
 530                // The TTL index is this store's ONLY cleanup mechanism (no application-side
 531                // pruning exists), so an operator-provisioned collection must be verified to
 532                // carry one — Cosmos and DynamoDB hard-fail the same way when their server-side
 533                // reaper is missing. Without this, a collection provisioned without
 534                // expireAfterSeconds grew without bound, with no error and no log line.
 535                VerifyTtlIndex(indexes!);
 536                _created = true;
 537                return;
 538            }
 539
 540            // A TTL index (expireAfterSeconds = 0 on the expiry timestamp) makes MongoDB itself
 541            // reap expired ledgers — no application-side pruning needed. Loads still filter on
 542            // ExpiresAtUtc because the TTL monitor only runs periodically (~60s).
 543            var indexName = $"{_options.CollectionName}_expires_idx";
 544            var model = new CreateIndexModel<MongoFlowStateDocument>(
 545                Builders<MongoFlowStateDocument>.IndexKeys.Ascending(item => item.ExpiresAtUtc),
 546                new CreateIndexOptions { Name = indexName, ExpireAfter = TimeSpan.Zero });
 547            // Do not drop or rewrite a conflicting application-owned index. MongoDB reports the
 548            // mismatch and startup fails, leaving the operator to correct schema intentionally —
 549            // unless what conflicts is the reaper this store needs under another name: the index
 550            // the docs and the AutoCreateIndexes = false error prescribe,
 551            // createIndex({ expires_at_utc: 1 }, { expireAfterSeconds: 0 }), is named
 552            // "expires_at_utc_1", and MongoDB refuses the same key and options under a second name
 553            // (85 IndexOptionsConflict; 86 IndexKeySpecsConflict for a same-named different key).
 554            try
 555            {
 556                await _collection.Indexes.CreateOneAsync(model, cancellationToken: cancellationToken).ConfigureAwait(fal
 557            }
 558            catch (MongoCommandException ex) when (ex.Code is 85 or 86)
 559            {
 560                indexes ??= await ListIndexesAsync(tolerateUnauthorized: false, cancellationToken).ConfigureAwait(false)
 561                if (!HasTtlIndex(indexes!))
 562                    throw;
 563            }
 564
 565            _created = true;
 566        }
 567        finally
 568        {
 569            _ensureGate.Release();
 570        }
 571    }

 598    /// <summary>
 599    /// Refuses a collection whose default collation folds: the flow id is the <c>_id</c>, and a
 600    /// collection created with a default collation builds its <c>_id_</c> index — which cannot be
 601    /// rebuilt — and evaluates every id equality under it. Two case- or accent-variant flow ids
 602    /// then collide: the second run's create reports the first run's ledger as existing, its load
 603    /// reads that ledger (refused as unreadable, so its start job dead-letters while the caller
 604    /// was told it started), and a delete removes the other run. Pinning the simple collation on
 605    /// the queries cannot help — the unique <c>_id_</c> index still folds — so startup refuses,
 606    /// as the relational stores refuse a folding <c>flow_id</c> column.
 607    /// </summary>
 608    private void ThrowIfFoldingCollation(List<BsonDocument> indexes)
 609    {
 610        var idIndex = indexes.Find(index => index.TryGetValue("name", out var name) && name == "_id_");
 611        if (idIndex is null
 612            || !idIndex.TryGetValue("collation", out var collation)
 613            || collation is not BsonDocument collationDocument
 614            || (collationDocument.TryGetValue("locale", out var locale) && locale == "simple"))
 615        {
 616            return;
 617        }
 618
 619        throw new InvalidOperationException(
 620            $"The MongoDB durable-flow collection '{_options.CollectionName}' was created with the default collation " +
 621            $"{collationDocument.ToJson()}, which its _id index — the flow id — carries too. The store compares flow ids
 622            "ORDINALLY, and a collation folds whatever its rules call equal — case, accents, or width, depending on the 
 623            "locale and strength — into one key. Distinct flow ids would then collide: the second run's create finds the
 624            "first run's ledger, its start job dead-letters, and a delete removes the other run. The _id index cannot be
 625            "rebuilt: recreate the collection without a collation (or with { locale: 'simple' }) and copy the documents 
 626            "over, or configure a new CollectionName.");
 627    }

 630    /// Verifies an operator-provisioned collection carries the TTL reaper this store depends on:
 631    /// a single-field index on the expiry timestamp with <c>expireAfterSeconds</c> set (any

 637            return;

 639        throw new InvalidOperationException(

 641            $"{nameof(MongoDbDurableFlowOptions)}.{nameof(MongoDbDurableFlowOptions.AutoCreateIndexes)} is disabled. The
 642            "the store's only cleanup mechanism; without it expired flow ledgers accumulate forever. Create it " +

 646    /// <summary>

 649    /// </summary>

 653            && index.TryGetValue("key", out var key)

 658    /// <summary>
 659    /// Server clock for the one write that cannot compute it in place: plain inserts evaluate no
 660    /// pipeline, so <c>$$NOW</c> is out of reach. <c>hello</c> is answered by every supported
 661    /// server (the 4.2+ floor the <c>$$NOW</c> filters already require) and carries the node's
 662    /// <c>localTime</c>; reading it from the primary keeps the authority the same node whose
 663    /// <c>$$NOW</c> the filters evaluate against.
 664    /// </summary>

 666    {
