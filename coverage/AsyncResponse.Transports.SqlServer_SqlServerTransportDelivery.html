 394    /// <summary>
 395    /// Whether a failure of the table-scoped application-lock step latches the retry-after window:
 396    /// only a lock wait lost — the batch's own "not granted" (<see cref="ApplicationLockFailed"/>)
 397    /// or a lock timeout (1222) — and never a transient connection fault (severity ≥ 20, a reset
 398    /// or aborted transport, an Azure failover error), which took nothing and is worth retrying at
 399    /// once.
 400    /// </summary>
 401    internal static bool LatchesOnTableLockFailure(SqlException ex)
 402        => ex.Class < 20 && ex.Number switch
 403        {
 404            // 1222 is itself in the transient set: it is the lock wait, not a connection fault.
 405            LockRequestTimeout => true,
 406            ApplicationLockFailed => !SqlServerTransientFaults.IsTransient(ex),
 407            _ => false,
 408        };

 507    /// <summary>
 508    /// The index builds' lock-wait bound, in milliseconds (PostgreSQL <c>DdlLockTimeout</c> parity):
 509    /// long enough to outwait the transport's own single-statement writes, and under the
 510    /// lease-renewal beat (<c>LockTimeout</c>/3, 10 s by default) so no running handler's renewal
 511    /// lapses behind a waiting build. On a genuinely busy table — a long transaction, an index
 512    /// rebuild, a bulk load holding a conflicting lock — the build fails fast instead, changing
 513    /// nothing, and <see cref="DdlLockTimeoutBackoff"/> paces the retries. It also bounds the wait
 514    /// for the table-scoped application lock another host holds while it builds.
 515    /// </summary>
 516    internal const int DdlLockTimeoutMilliseconds = 5000;
 517
 518    /// <summary>
 519    /// The shortest retry-after window a failed startup-DDL attempt latches; the actual window is
 520    /// jittered up to twice this (30–60 s) — PostgreSQL <c>RetryAfter</c> parity. Without it every
 521    /// operation on the host started the next attempt at once, and a lock held for minutes stalled
 522    /// the whole queue — every host's writes, queued behind each attempt's lock request — in
 523    /// back-to-back 5 s cycles, the k-th caller queued on the gate waiting k × 5 s to fail, and a
 524    /// build that failed for another reason re-ran, write block and all, on every operation. 30 s
 525    /// keeps one host's attempts to a small fraction of the table's time; the jitter keeps a fleet
 526    /// started together from retrying in lockstep (the application lock would chain their waits back
 527    /// to back), and the 60 s ceiling bounds how long this host stays unable to publish or claim once
 528    /// the cause is gone.
 529    /// </summary>
 530    internal static readonly TimeSpan DdlLockTimeoutBackoff = TimeSpan.FromSeconds(30);
 531
 532    /// <summary>
 533    /// The command timeout of DDL whose run time grows with the table — the first build of an index
 534    /// on an existing one — in seconds (PostgreSQL parity). Not SqlClient's 30 s default: the build
 535    /// is one-time, and one that outran it rolled back and was retried by every later operation,
 536    /// blocking writes for another full timeout each time and never finishing. Not unbounded either:
 537    /// a connection that never answers would leave every later operation on the host waiting behind
 538    /// it. An hour is an order of magnitude past what a queue table's index build takes; a larger
 539    /// table gets the index ahead of the rollout (docs/sqlserver.md).
 540    /// </summary>
 541    internal const int LongRunningDdlCommandTimeoutSeconds = 3600;

 543    /// <summary>
 544    /// A command for DDL whose run time grows with the table (see

 551    /// <summary>
 552    /// Latches this store's retry-after window after the startup DDL lost a lock wait (see
 553    /// <see cref="DdlLockTimeoutBackoff"/>) and returns its length. Until it ends,
 554    /// <see cref="EnsureCreatedAsync"/> fails at once — no connection, no transaction, no lock
 555    /// request queued ahead of the other hosts' writes.
 556    /// </summary>
 557    internal TimeSpan BackOffDdlAfterLockTimeout(Exception cause) => BackOffDdl(cause, lockWait: true);

 560    {
 561        var window = DdlLockTimeoutBackoff + TimeSpan.FromTicks(Random.Shared.NextInt64(DdlLockTimeoutBackoff.Ticks));
 562        Volatile.Write(ref _ddlBackoff, new DdlBackoff(Clock.GetTimestamp(), window, cause, lockWait));
 563        return window;
 564    }

 674    /// leaves behind — and by a disabled index, which serves nothing.
 675    /// </summary>
 676    internal const string DequeueIndexQuery =
 677        """
 678        SELECT CASE WHEN EXISTS (
 679            SELECT 1
 680            FROM sys.indexes i
 681            WHERE i.object_id = OBJECT_ID(@table)
 682              AND i.is_disabled = 0 AND i.is_hypothetical = 0 AND i.has_filter = 0
 683              AND (SELECT COUNT(*)
 684                   FROM sys.index_columns ic
 685                   JOIN sys.columns c ON c.object_id = ic.object_id AND c.column_id = ic.column_id
 686                   WHERE ic.object_id = i.object_id AND ic.index_id = i.index_id AND ic.is_descending_key = 0
 687                     AND ((ic.key_ordinal = 1 AND c.name = N'queue')
 688                       OR (ic.key_ordinal = 2 AND c.name = N'available_at')
 689                       OR (ic.key_ordinal = 3 AND c.name = N'created_at'))) = 3) THEN 1 ELSE 0 END;
 690        """;

 773    /// <summary>
 774    /// Whether a publish raises <see cref="MessagePublished"/>: only when its row is claimable at

 777    /// every durable-flow timer park and redelay hop and every delayed enqueue. The row is picked up
 778    /// by the first poll tick after it falls due.

 782    public async Task<SqlServerTransportDelivery?> TryClaimAsync(string queue, TimeSpan lockTimeout, CancellationToken c
 783    {
 784        await EnsureCreatedAsync(cancellationToken).ConfigureAwait(false);
 785        var lockId = Guid.NewGuid();
 786
 787        await using var connection = await OpenConnectionAsync(cancellationToken).ConfigureAwait(false);
 788        await using var command = connection.CreateCommand();
 789        // READPAST skips rows other subscribers hold UPDLOCK on — SQL Server's equivalent of
 790        // PostgreSQL's FOR UPDATE SKIP LOCKED — so competing consumers never block on each other.
 791        // Ready order is AVAILABILITY order — (available_at, created_at), the dequeue index's own
 792        // key order behind the queue equality — so the scan returns its first unleased row without
 793        // sorting. It equals publish order for every row that was neither delayed nor NAKed (both
 794        // columns default to the same SYSUTCDATETIME()); a delayed or redelivered row queues by
 795        // when it became due instead of jumping ahead of everything published while it waited.
 796        command.CommandText =
 797            $"""
 798            WITH next AS (
 799                SELECT TOP (1) id, payload_json, headers_json, attempts, locked_until, lock_id
 800                FROM {MessageTable} WITH (UPDLOCK, ROWLOCK, READPAST)
 801                WHERE {ExactQueueMatch}
 802                  AND available_at <= SYSUTCDATETIME()
 803                  AND (locked_until IS NULL OR locked_until <= SYSUTCDATETIME())
 804                ORDER BY available_at, created_at
 805            )
 806            UPDATE next
 807            SET attempts = attempts + 1,
 808                locked_until = {AddMilliseconds("@lock_timeout_ms")},
 809                lock_id = @lock_id
 810            OUTPUT inserted.id, inserted.payload_json, inserted.headers_json, inserted.attempts;
 811            """;
 812        command.Parameters.AddWithValue("@queue", queue);
 813        command.Parameters.AddWithValue("@lock_timeout_ms", (long)lockTimeout.TotalMilliseconds);
 814        command.Parameters.AddWithValue("@lock_id", lockId);
 815
 816        await using var reader = await command.ExecuteReaderAsync(cancellationToken).ConfigureAwait(false);
 817        if (!await reader.ReadAsync(cancellationToken).ConfigureAwait(false))
 818            return null;
 819
 820        var id = reader.GetGuid(0);
 821        var payload = reader.GetString(1);
 822        var headerJson = reader.GetString(2);
 823        var attempt = reader.GetInt32(3);
 824        var headers = DeserializeHeaders(headerJson);
 825
 826        // The claim predicate matches the queue exactly (see ExactQueueMatch), so the claimed row's
 827        // queue IS the requested one — no post-claim re-check, and therefore no row that gets
 828        // claimed, rejected, and released back to the head of the same ordering on every poll.
 829        return new SqlServerTransportDelivery(
 830            id,
 831            queue,
 832            payload,
 833            headers,
 834            attempt,
 835            () => AckAsync(id, lockId),
 836            delay => NakAsync(id, lockId, delay),
 837            (exception, deleteOriginal, token) => DeadLetterAsync(id, lockId, queue, payload, headers, exception, delete
 838            token => RenewLeaseAsync(id, lockId, lockTimeout, token));
 839    }

 858        string queue,
 859        string payload,

 861        string? deadLetterReason,

 875            delay is null

 878                  SELECT @id, @queue, @payload_json, @headers_json, @dead_letter_reason
 879                  WHERE NOT EXISTS (SELECT 1 FROM {MessageTable} WITH (UPDLOCK, HOLDLOCK) WHERE id = @id);
 880                  """
 881                : $"""
 882                  INSERT INTO {MessageTable} (id, queue, payload_json, headers_json, dead_letter_reason, available_at)
 883                  SELECT @id, @queue, @payload_json, @headers_json, @dead_letter_reason, {AddMilliseconds("@available_de
 884                  WHERE NOT EXISTS (SELECT 1 FROM {MessageTable} WITH (UPDLOCK, HOLDLOCK) WHERE id = @id);
 885                  """;

 887        command.Parameters.AddWithValue("@queue", queue);

 890        command.Parameters.AddWithValue("@dead_letter_reason", (object?)deadLetterReason ?? DBNull.Value);

 895        {
 896            await command.ExecuteNonQueryAsync(cancellationToken).ConfigureAwait(false);
 897        }
 898        catch (SqlException ex) when (ex.Number is PrimaryKeyViolation or UniqueIndexViolation)
 899        {

 901
 902        if (notify)
 903            MessagePublished?.Invoke(queue);
 904    }

 906    private async ValueTask AckAsync(Guid id, Guid lockId)
 907    {
 908        await using var connection = await OpenConnectionAsync(CancellationToken.None).ConfigureAwait(false);
 909        await using var command = connection.CreateCommand();
 910        command.CommandText = $"DELETE FROM {MessageTable} WHERE id = @id AND lock_id = @lock_id;";
 911        command.Parameters.AddWithValue("@id", id);
 912        command.Parameters.AddWithValue("@lock_id", lockId);
 913        await command.ExecuteNonQueryAsync(CancellationToken.None).ConfigureAwait(false);
 914    }
 915
 916    // The token is the heartbeat's per-attempt bound: a renew hung on a black-holed pooled
 917    // connection, a failover, or a lock wait inherited SqlClient's 30 s command timeout, which
 918    // outlasted the 20 s of lease left after the beat — the lease lapsed before the short-backoff
 919    // retry ever ran. CommandTimeout is the backstop for a cancellation whose attention packet
 920    // itself hangs on a dead socket.
 921    private async ValueTask<bool> RenewLeaseAsync(Guid id, Guid lockId, TimeSpan lockTimeout, CancellationToken cancella
 922    {
 923        await using var connection = await OpenConnectionAsync(cancellationToken).ConfigureAwait(false);
 924        await using var command = connection.CreateCommand();
 925        command.CommandTimeout = RenewCommandTimeoutSeconds(lockTimeout);
 926        // The row count comes from the SERVER (@@ROWCOUNT, which SET NOCOUNT does not affect), not
 927        // from ExecuteNonQuery: under a server-wide `user options` NOCOUNT (sp_configure 512) that
 928        // returns -1, every successful renew read as "lease lost", the heartbeat stopped after its
 929        // first beat, and a handler longer than the lease ran twice concurrently. (The dead-letter
 930        // batch below reads its outcome the same way.)
 931        command.CommandText =
 932            $"""
 933            UPDATE {MessageTable}
 934            SET locked_until = {AddMilliseconds("@lock_timeout_ms")}
 935            WHERE id = @id AND lock_id = @lock_id;
 936            SELECT @@ROWCOUNT;
 937            """;
 938        command.Parameters.AddWithValue("@id", id);
 939        command.Parameters.AddWithValue("@lock_id", lockId);
 940        command.Parameters.AddWithValue("@lock_timeout_ms", (long)lockTimeout.TotalMilliseconds);
 941        return await command.ExecuteScalarAsync(cancellationToken).ConfigureAwait(false) is int renewed && renewed > 0;
 942    }

 962    private async ValueTask<bool> DeadLetterAsync(
 963        Guid id,
 964        Guid lockId,
 965        string sourceQueue,
 966        string payload,
 967        IReadOnlyDictionary<string, string> headers,
 968        Exception exception,
 969        bool deleteOriginal,
 970        CancellationToken cancellationToken)
 971    {
 972        if (!_options.DeadLetterEnabled)
 973        {
 974            if (deleteOriginal)
 975                await AckAsync(id, lockId).ConfigureAwait(false);
 976            return true;
 977        }
 978
 979        var deadHeaders = new Dictionary<string, string>(headers, StringComparer.OrdinalIgnoreCase)
 980        {
 981            ["AR-DeadLetter-Reason"] = Sanitize(exception.Message),
 982            ["AR-DeadLetter-Source-Queue"] = sourceQueue
 983        };
 984
 985        try
 986        {
 987            if (!deleteOriginal)
 988            {
 989                await InsertAsync(Guid.NewGuid(), _options.DeadLetterQueue, payload, deadHeaders, exception.Message, not
 990                return true;
 991            }
 992
 993            // The DLQ insert and the original-row delete must commit atomically: split across two
 994            // connections, a crash between them leaves the original row to be redelivered and
 995            // dead-lettered again, duplicating the DLQ entry.
 996            await EnsureCreatedAsync(cancellationToken).ConfigureAwait(false);
 997            await using var connection = await OpenConnectionAsync(cancellationToken).ConfigureAwait(false);
 998            await using var transaction = (SqlTransaction)await connection.BeginTransactionAsync(cancellationToken).Conf
 999            await using var command = connection.CreateCommand();
 1000            command.Transaction = transaction;
 1001            // Delete FIRST and write the DLQ row only if the fence matched. A stale claim (the lease
 1002            // lapsed and a peer re-claimed the row) must no-op here exactly as the fenced ack and
 1003            // NAK do; writing the row unconditionally buried a full copy of a message that is still
 1004            // live and may yet succeed under its new owner, so the DLQ showed a poison entry for
 1005            // work that completed — and an operator replaying it duplicated its side effects.
 1006            command.CommandText =
 1007                $"""
 1008                SET NOCOUNT ON;
 1009                DELETE FROM {MessageTable} WHERE id = @source_id AND lock_id = @lock_id;
 1010                IF @@ROWCOUNT = 1
 1011                BEGIN
 1012                    INSERT INTO {MessageTable} (id, queue, payload_json, headers_json, dead_letter_reason)
 1013                    VALUES (@id, @queue, @payload_json, @headers_json, @dead_letter_reason);
 1014                    SELECT 1;
 1015                END
