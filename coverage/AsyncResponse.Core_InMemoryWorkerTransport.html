 990/// queues <c>ExecuteAsync</c> to the thread pool and DISCARDS the queued work when the stopping

 996/// synchronously inside <see cref="StartAsync"/>, before it returns.
 997/// </para>
 998/// </summary>

 1067    private async Task RunWorkerAsync(CancellationToken stoppingToken)
 1068    {
 1069        // Off the caller's thread before the first read. A queue that already holds jobs (published
 1070        // before the host started, e.g. by an earlier hosted service's StartAsync) completes the

 1073        // backlog, spent its startup timeout on it, and deadlocked when a handler waited on a
 1074        // service started later. The drain hook is already installed by RunAsync. Onto the thread
 1075        // pool, never the caller's SynchronizationContext (which Task.Yield posts to): a host
 1076        // started on a UI thread, or under any single-threaded context, ran no worker until that
 1077        // thread pumped — and deadlocked if it then blocked on a job's result.
 1078        await Task.CompletedTask.ConfigureAwait(ConfigureAwaitOptions.ForceYielding);
 1079
 1080        // Deliberately no cancellation token on the read: the loop ends when the completed queue
 1081        // is empty, never by abandoning accepted jobs mid-queue.
 1082        await foreach (var queued in _transport.Reader.ReadAllAsync().ConfigureAwait(false))
 1083        {
 1084            // Counted as started under the pump gate — or held back for a simulated restart.
 1085            if (!_transport.TryStartJob(queued))
 1086                continue;

 1088            await RunJobAsync(queued, stoppingToken).ConfigureAwait(false);
 1089
 1090            // Follow-up work first: whatever the pump could not place in the queue is run here,
 1091            // BEFORE the next read — the read is what hands the freed slot to a parked external
 1092            // producer, and it would keep doing so for as long as one is waiting (see
 1093            // TryTakeOverflow). Each follow-up is an outstanding job in its own right and is
 1094            // finished through the same path as a queued one, pump included — so whenever the
 1095            // queue does have room, idle workers share a large fan-out instead of this worker
 1096            // running all of it serially.
 1097            //
 1098            // Bounded by the overflow present when the burst starts: a follow-up's own
 1099            // follow-up finds the queue just as full (nothing has read it), so an unbounded burst
 1100            // ran a whole chain — a sequential child orchestration, a job that re-enqueues
 1101            // itself — before any job already waiting in the queue, for as long as the chain
 1102            // kept going. What the burst adds waits for the next one, one queue read later:

 1104            for (var burst = _transport.OverflowDepth; burst > 0 && _transport.TryTakeOverflow(out var followUp); burst-

 1106        }

 1283    private static TimeSpan RetryDelay(InMemoryWorkerTransportOptions options, int attempt)
 1284    {

 1286        // a pre-shift comparison so neither pathological attempt counts (unlimited retries) nor a
 1287        // pathological base delay can overflow before the saturation check.
 1288        var exponent = Math.Min(attempt - 1, 20);
 1289        var baseTicks = options.RetryBaseDelay.Ticks;
 1290        var maxTicks = options.RetryMaxDelay.Ticks;
 1291        return baseTicks > maxTicks >> exponent
 1292            ? options.RetryMaxDelay
 1293            : TimeSpan.FromTicks(baseTicks << exponent);
 1294    }
