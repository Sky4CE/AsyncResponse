 448    /// <summary>

 458    private async Task<(string Owner, string Name)?> ResolveBaseTableAsync(OracleConnection connection, CancellationToke
 459    {

 470        var name = CatalogName;

 564    /// <summary>
 565    /// Requires ordinal NVARCHAR2 comparison in this session. NLS_COMP=BINARY (Oracle's default)
 566    /// compares bytes regardless of NLS_SORT; NLS_COMP=LINGUISTIC (or the deprecated ANSI) routes
 567    /// every comparison through NLS_SORT instead, where anything but BINARY — BINARY_CI,
 568    /// BINARY_AI, a language sort — folds case or accents. Sessions inherit these from instance
 569    /// parameters, client NLS configuration, and logon triggers, uniformly for every connection
 570    /// this store opens, so one check on one pooled session stands for all of them. Internal (not
 571    /// private) so the integration suite can run it against a deliberately mis-set session without
 572    /// installing a logon trigger.
 573    /// </summary>
 574    internal static async Task VerifyComparisonSemanticsAsync(OracleConnection connection, CancellationToken cancellatio
 575    {
 576        string? comp = null;
 577        string? sort = null;
 578        await using (var command = connection.CreateCommand())
 579        {
 580            command.CommandText = "SELECT PARAMETER, VALUE FROM NLS_SESSION_PARAMETERS WHERE PARAMETER IN ('NLS_COMP', '
 581            await using var reader = await command.ExecuteReaderAsync(cancellationToken).ConfigureAwait(false);
 582            while (await reader.ReadAsync(cancellationToken).ConfigureAwait(false))
 583            {
 584                if (string.Equals(reader.GetString(0), "NLS_COMP", StringComparison.OrdinalIgnoreCase))
 585                    comp = reader.GetString(1);
 586                else
 587                    sort = reader.GetString(1);
 588            }
 589        }
 590
 591        if (DiagnoseComparisonSemantics(comp, sort) is { } linguisticSession)
 592            throw new InvalidOperationException(linguisticSession);
 593    }

 595    /// <summary>
 596    /// The comparison-semantics decision on its own catalog inputs: the rejection message for a
 597    /// session whose NLS settings fold flow-id equality, or null for an ordinal one. Folded
 598    /// equality is the silent kind of wrong — every <c>WHERE flow_id = :flow_id</c> this store
 599    /// runs matches BOTH casings, so a load returns another flow's state and a lease update fences
 600    /// the other flow's execution, while the primary-key index (always binary) keeps admitting
 601    /// both rows. Nothing errors; flows cross-route. A session whose parameters could not be read
 602    /// is not known to be ordinal, so silence does not pass.
 603    /// </summary>
 604    internal static string? DiagnoseComparisonSemantics(string? nlsComp, string? nlsSort)
 605    {
 606        static bool IsBinary(string? value) => string.Equals(value, "BINARY", StringComparison.OrdinalIgnoreCase);
 607        if (IsBinary(nlsComp) || IsBinary(nlsSort))
 608            return null;
 609
 610        return
 611            $"This Oracle session compares text linguistically (NLS_COMP='{nlsComp ?? "(unknown)"}', " +
 612            $"NLS_SORT='{nlsSort ?? "(unknown)"}'), so flow ids differing only in case (or accent) match each " +
 613            "other's rows: a load can return the other flow's state and a lease can fence the other flow's " +
 614            "execution, while the binary primary-key index still admits both ids. Flow ids are compared " +
 615            "ordinally by the engine. Restore binary comparison with ALTER SESSION SET NLS_COMP = BINARY (or " +
 616            "NLS_SORT = BINARY) — typically by removing the logon trigger or client NLS configuration that " +
 617            "changed them.";
 618    }

 624    internal readonly record struct ActualColumn(
 625        string DataType,
 626        bool Nullable,
 627        long? CharLength,
 628        long? Precision,
 629        long? Scale,
 630        bool HasDefault,
 631        bool Virtual,
 632        bool Identity)
 633    {
 634        /// <summary>
 635        /// Whether this store could insert a row without naming this column. True when the column
 636        /// is nullable, carries a default, is computed by the database, or is an identity.
 637        /// </summary>
 638        internal bool IsWritableWithoutValue => Nullable || HasDefault || Virtual || Identity;
 639    }

 641    /// <summary>
 642    /// What this store needs from one column, and the check that says so. Widths and precisions
 643    /// are MINIMA rather than exact matches: a wider flow_id or a higher-precision timestamp still
 644    /// satisfies every promise the store makes, and rejecting a more generous schema would be a
 645    /// false alarm. Too NARROW is not — NVARCHAR2(10) passes a name-only check and then errors on
 646    /// the first 400-character id the public contract permits.
 647    /// </summary>
 648    internal sealed record ExpectedColumn(string Name, string Declaration, string DataType, bool Nullable, long? Minimum
 649    {
 650        internal string? Mismatch(ActualColumn actual)
 651        {
 652            if (string.Equals(DataType, "TIMESTAMP", StringComparison.Ordinal))
 653            {
 654                // USER_TAB_COLS embeds the fractional precision in the type name ('TIMESTAMP(6)'),
 655                // so the family is a prefix match. The WITH [LOCAL] TIME ZONE variants are
 656                // rejected as different types: their values shift with the session time zone, and
 657                // expiry/lease math must stay on the plain UTC timestamps this store writes.
 658                if (!actual.DataType.StartsWith("TIMESTAMP", StringComparison.OrdinalIgnoreCase)
 659                    || actual.DataType.Contains("TIME ZONE", StringComparison.OrdinalIgnoreCase))
 660                {
 661                    return $"is a '{actual.DataType}'";
 662                }
 663            }
 664            else if (!string.Equals(actual.DataType, DataType, StringComparison.OrdinalIgnoreCase))
 665            {
 666                return $"is a '{actual.DataType}'";
 667            }
 668
 669            if (actual.Nullable != Nullable)
 670                return Nullable ? "is NOT NULL (this store writes NULL to it)" : "is nullable";
 671            if (Minimum is not { } minimum)
 672                return null;
 673
 674            // CHAR_LENGTH for the character columns, DATA_SCALE (fractional-second digits) for the
 675            // timestamps, DATA_PRECISION for NUMBER: a TIMESTAMP(0) column silently rounds the
 676            // sub-second lease arithmetic this store runs on SYS_EXTRACT_UTC(SYSTIMESTAMP), which
 677            // is how two workers end up holding one lease, and a NUMBER(9) revision overflows
 678            // without a word. An unconstrained NUMBER (null precision) holds 38 digits and passes.
 679            var actualSize = DataType switch
 680            {
 681                "TIMESTAMP" => actual.Scale,
 682                "NUMBER" => actual.Precision ?? 38,
 683                _ => actual.CharLength
 684            };
