 244    private async Task<T> WithCancellation<T>(Task<T> command, CancellationToken cancellationToken)
 245    {
 246        // StackExchange.Redis enforces its own sync/async command timeouts; this adds an upper bound that
 247        // also honors the caller's token (e.g. host shutdown). On timeout the in-flight command is
 248        // abandoned best-effort — the multiplexer keeps running it — and surfaced as a TimeoutException so
 249        // the retry paths treat it as transient, while a genuine caller cancellation stays an
 250        // OperationCanceledException and is not retried.
 251        using var timeout = new CancellationTokenSource(_operationTimeout);
 252        using var linked = CancellationTokenSource.CreateLinkedTokenSource(cancellationToken, timeout.Token);
 253        try
 254        {
 255            return await command.WaitAsync(linked.Token).ConfigureAwait(false);
 256        }
 257        catch (OperationCanceledException) when (timeout.IsCancellationRequested && !cancellationToken.IsCancellationReq
 258        {
 259            throw new TimeoutException($"The Redis command did not complete within {_operationTimeout}.");
 260        }
