 688    /// <summary>The longest well-formed prefix of <paramref name="value"/> that encodes to at most <paramref name="maxB
 689    private static string TruncateToUtf8Bytes(string value, long maxBytes)
 690    {
 691        if (maxBytes <= 0)
 692            return string.Empty;
 693        if (Encoding.UTF8.GetByteCount(value) <= maxBytes)
 694            return value;
 695
 696        // Every UTF-16 unit encodes to at least one byte, so the answer is at most maxBytes units.
 697        int low = 0, high = (int)Math.Min(value.Length, maxBytes);
 698        while (low < high)
 699        {
 700            var mid = low + ((high - low + 1) / 2);
 701            if (Encoding.UTF8.GetByteCount(PortableText.TruncateWellFormed(value, mid)) <= maxBytes)
 702                low = mid;
 703            else
 704                high = mid - 1;
 705        }
 706
 707        return PortableText.TruncateWellFormed(value, low);
 708    }

 710    private async Task<bool> DeadLetterCoreAsync(
 711        string sourceTopic,
 712        int partition,
 713        long offset,
 714        byte[] payload,
 715        IReadOnlyList<KafkaTransportHeader> originalHeaders,
 716        string? correlationId,
 717        Exception exception,
 718        string reason,
 719        int attempts,
 720        CancellationToken cancellationToken)
 721    {
 722        if (!TransportOptions.DeadLetterEnabled)
 723            return false;
 724
 725        // A record replayed from a dead-letter topic already carries a burial set: replaced, not
 726        // stacked — a second sourceTopic/reason/... behind the first left every first-match reader
 727        // (tooling, a human with kcat) looking at the stale values, and each replay cycle grew the
 728        // record toward the size limit. Only a record that carries one (sourceTopic AND
 729        // sourceOffset) is stripped: the names are generic, and a user's own "reason" or
 730        // "attempts" header on a record that was never buried is part of the message.
 731        var replayedBurial = HasHeader(originalHeaders, "sourceTopic") && HasHeader(originalHeaders, "sourceOffset");
 732        var headers = new List<KafkaTransportHeader>(originalHeaders.Count + BurialHeaderKeys.Count);
 733        foreach (var header in originalHeaders)
 734        {
 735            if (!replayedBurial || !BurialHeaderKeys.Contains(header.Key))
 736                headers.Add(header);
 737        }
 738
 739        headers.Add(KafkaTransportHeader.Utf8("sourceTopic", sourceTopic));
 740        headers.Add(KafkaTransportHeader.Utf8("sourcePartition", partition.ToString(CultureInfo.InvariantCulture)));
 741        headers.Add(KafkaTransportHeader.Utf8("sourceOffset", offset.ToString(CultureInfo.InvariantCulture)));
 742        headers.Add(KafkaTransportHeader.Utf8("consumerGroup", _consumerGroup));
 743        headers.Add(KafkaTransportHeader.Utf8("subscriberRole", _role.ToString()));
 744        headers.Add(KafkaTransportHeader.Utf8("attempts", attempts.ToString(CultureInfo.InvariantCulture)));
 745        headers.Add(KafkaTransportHeader.Utf8("reason", reason));
 746        headers.Add(KafkaTransportHeader.Utf8("occurredAtUtc", DateTimeOffset.UtcNow.ToString("O")));
 747
 748        // Capped: these two are the only dead-letter headers whose size the failing code decides
 749        // (an exception message can quote a whole payload; a closed generic's name nests without
 750        // bound), and an uncapped one pushed the dead-letter record past message.max.bytes — a
 751        // burial that then fails on every retry, for a message that was itself within the limit.
 752        // Surrogate-aware cut (see PortableText.TruncateWellFormed). The per-header cap alone did
 753        // not close that: a record within ~24 KB of the producer's limit still could not be buried
 754        // (and restarted the subscriber for ever, see KafkaDeadLetterPublishFailedException), so
 755        // both are also cut to what the record's own size leaves — the message first.

 881/// <summary>
 882/// Ack-after-handler mode. A message's handler is started the moment it is consumed and awaited
 883/// inline for up to <see cref="KafkaSubscriberOptions.DetachHandlerAfter"/>; a handler still
 884/// running past that is <em>detached</em>: its partition is paused (Kafka's own ordering primitive
 885/// — nothing for it is fetched, nothing is buffered in-process), the handler and its retry ladder
 886/// run on, and the poll thread returns to polling. The earlier design awaited the whole handler on
 887/// the poll thread: a durable-flow step awaiting a remote response or a timer for longer than
 888/// <c>max.poll.interval.ms</c> (5 minutes by default) got the consumer evicted from its group, its
 889/// partitions rebalanced, the message redelivered to a peer that started the same work again,
 890/// and every other partition assigned to this consumer stalled behind it.
 891/// <para>
 892/// The consumer is touched only from the poll thread: detached handlers never store offsets or
 893/// resume partitions themselves. The poll loop calls <see cref="SettleCompleted"/> every tick,
 894/// which observes finished handlers exactly as the inline path would — success stores the offset,
 895/// cancellation leaves it unstored for redelivery, a burial that failed for good faults the poll
 896/// loop so nothing is ever committed past the message — then starts the next message held for the
 897/// partition, or resumes it. Disposal (the poll loop has exited by then) waits for the remaining
 898/// detached handlers and settles them before the consumer's close commits, so finished work is
