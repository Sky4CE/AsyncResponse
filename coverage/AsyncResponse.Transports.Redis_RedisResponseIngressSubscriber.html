 368        // XPENDING still lists up to PendingClaimBatchSize candidates, so reclaim throughput does
 369        // not collapse to one entry per PendingClaimInterval — but each is XCLAIMed only right
 370        // before it runs, so only the entry about to execute has its count bumped. XCLAIM's
 371        // min-idle re-check skips a candidate a peer took in the meantime.
 372        var processed = 0;
 373        foreach (var candidate in pending)
 374        {
 375            // Stopping: claim nothing more — an unclaimed candidate keeps its count and stays
 376            // claimable by a peer.
 377            if (cancellationToken.IsCancellationRequested || TakesNothingMore(dispatcher))
 378                break;
 379
 380            // The listing is as old as every earlier candidate's handler run — minutes, behind a
 381            // slow one — and a peer may have claimed, failed and released this entry since, so its
 382            // delivery count there is stale: the attempt number would come out one low per missed
 383            // claim, running a poison entry past MaxDeliveryAttempts and dead-lettering it with a
 384            // wrong attempt. Re-read just this id, which also re-checks that it is still idle
 385            // enough to claim; gone from the listing means a peer holds it (or it was settled).
 386            var current = await _database.StreamPendingMessagesAsync(
 387                Stream,
 388                ConsumerGroup,
 389                1,
 390                RedisValue.Null,
 391                minId: candidate.MessageId,
 392                maxId: candidate.MessageId,
 393                minIdleMs,
 394                cancellationToken).ConfigureAwait(false);
 395            if (current.Length == 0)
 396                continue;
 397
 398            // The re-read took a round trip: a stop landing in it must not still claim (the
 399            // adapter would queue the XCLAIM with the cancelled token, bumping the count of an
 400            // entry that then sits in this stopping consumer's pending list).
 401            if (cancellationToken.IsCancellationRequested || TakesNothingMore(dispatcher))
 402                break;
 403
 404            var result = await ClaimAndDispatchAsync(dispatcher, consumerName, current, minIdleMs, cancellationToken).Co
 405            processed += result.Processed;
 406
 407            // The flow engine handed a delivery back: the host is stopping, even if this
 408            // subscriber's token has not been cancelled yet. Treated like the stop itself.
 409            if (result.HandedBack)
 410                break;
 411        }
 412
 413        return processed;
 414    }

 416    private async Task<BatchResult> ClaimAndDispatchAsync(
 417        RedisMessageDispatcher dispatcher,
 418        RedisValue consumerName,
 419        StreamPendingMessageInfo[] pending,
 420        long minIdleMs,
 421        CancellationToken cancellationToken)
 422    {
 423        var pendingById = pending.ToDictionary(
 424            item => item.MessageId.ToString(),
 425            StringComparer.Ordinal);
 426        var claimed = await _database.StreamClaimAsync(
 427            Stream,
 428            ConsumerGroup,
 429            consumerName,
 430            minIdleMs,
 431            pending.Select(item => item.MessageId).ToArray(),
 432            cancellationToken).ConfigureAwait(false);
 433
 434        // Redis 6.2 answers XCLAIM with a nil entry for an id whose message was trimmed while still
 435        // pending (7.x drops it from the PEL instead). A nil entry has no id, so neither the
 436        // dispatch path nor the JUSTID heartbeat can name it, and it stayed in the PEL to be
 437        // re-claimed every cycle. When the reply is complete its order matches the request, so
 438        // the tombstones are ACKed by their pending ids here; a partial reply leaves them for the
 439        // next cycle. Either way only real entries are dispatched.
 440        if (Array.Exists(claimed, static entry => entry.Id.IsNull))
 441        {
 442            if (claimed.Length == pending.Length)
 443            {
 444                for (var index = 0; index < claimed.Length; index++)
 445                {
 446                    if (!claimed[index].Id.IsNull)
 447                        continue;
 448
 449                    var tombstoneId = pending[index].MessageId;
 450                    try
 451                    {
 452                        await _database.StreamAcknowledgeAsync(Stream, ConsumerGroup, tombstoneId, CancellationToken.Non
 453                        Logger.LogWarning(
 454                            "Redis pending entry {MessageId} on {Stream} was trimmed while still pending; ACKed the tomb
 455                            tombstoneId.ToString(),
 456                            Stream.ToString());
 457                    }
 458                    catch (Exception ex)
 459                    {
 460                        Logger.LogWarning(
 461                            ex,
 462                            "Failed to ACK trimmed pending entry {MessageId} on {Stream}; it is retried on the next pend
 463                            tombstoneId.ToString(),
 464                            Stream.ToString());
 465                    }
 466                }
 467            }
 468
 469            claimed = Array.FindAll(claimed, static entry => !entry.Id.IsNull);
 470        }
 471
 472        return await DispatchBatchAsync(
 473            dispatcher,
 474            claimed,
 475            consumerName,
 476            entry =>
 477            {
 478                var priorDeliveries = pendingById.TryGetValue(entry.Id.ToString(), out var info)
 479                    ? info.DeliveryCount
 480                    : 1;
 481                return Math.Max(1, priorDeliveries + 1);
 482            },
 483            cancellationToken).ConfigureAwait(false);
 484    }

 486    /// <summary>What a dispatched batch came to: the entries that counted as progress, and whether the flow engine hand
 487    private readonly record struct BatchResult(int Processed, bool HandedBack);
 488
 489    private async Task<BatchResult> DispatchBatchAsync(
 490        RedisMessageDispatcher dispatcher,
 491        StreamEntry[] entries,
 492        RedisValue consumerName,
 493        Func<StreamEntry, int> attemptFor,
 494        CancellationToken stoppingToken)
 495    {
 496        if (entries.Length == 0)
 497            return default;
 498
 499        // The batch is dispatched serially, and XREADGROUP/XCLAIM stamped every entry's idle
 500        // clock at read time — so a slow handler lets the idle time of the later (still
 501        // unprocessed) entries cross PendingMessageMinIdleTime, where a sibling's pending-claim
 502        // scan steals and re-runs them concurrently, bumping their PEL delivery count toward the
 503        // dead-letter cap on work that never once failed. While the batch is in flight, a
 504        // heartbeat claims the unprocessed entries back to this consumer with XCLAIM JUSTID,
 505        // which resets idle WITHOUT bumping the delivery count. The heartbeat is NOT tied to the
 506        // stop token: a handler takes no token, so it outlives the stop signal, and cancelling its
 507        // idle reset there let PendingMessageMinIdleTime lapse under the live handler on every
 508        // rolling deploy — a peer's pending claim took the entry and ran it a second time while it
 509        // was still executing here. It ends only when this loop has let go of the batch.
 510        var progress = new BatchProgress();
 511        using var renewalCancellation = new CancellationTokenSource();
 512        var renewalTask = RenewClaimLoopAsync(entries, consumerName, progress, stoppingToken, renewalCancellation.Token)
 513        var processed = 0;
 514        var handedBack = false;
 515        try
 516        {
 517            foreach (var entry in entries)
 518            {
 519                // Stopping: do not start what has not started. The rest of the batch used to run
 520                // on, handler after handler, past the stop signal; left unstarted it stays pending
 521                // (Redis has no NAK), its idle clock no longer reset, for a peer to reclaim. A
 522                // hand-back from the flow engine — here, or in an early-ACK worker — is the same
 523                // signal, arriving before the token, and so is the worker's intake gate closing at
 524                // ApplicationStopping: an early-ACK entry is never enqueued and ACKed after it.
 525                if (stoppingToken.IsCancellationRequested || handedBack || TakesNothingMore(dispatcher))
 526                    break;
 527
 528                try
 529                {
 530                    switch (await DispatchEntryAsync(dispatcher, entry, attemptFor(entry), stoppingToken).ConfigureAwait
 531                    {
 532                        case RedisDispatchOutcome.Processed:
 533                            processed++;
 534                            break;
 535                        case RedisDispatchOutcome.HandedBack:
 536                            handedBack = true;
 537                            break;
 538                    }
 539                }
 540                finally
 541                {
 542                    // Also counts Deferred and handed-back entries: they were left pending ON
 543                    // PURPOSE, so the heartbeat must stop touching them and let their idle accrue
 544                    // toward reclaim.
 545                    progress.MarkSettled();
 546                }
 547            }
 548        }
 549        finally
 550        {
 551            renewalCancellation.Cancel();
 552            await renewalTask.ConfigureAwait(false);
 553        }
 554
 555        return new BatchResult(processed, handedBack);
 556    }

 558    private async Task RenewClaimLoopAsync(
 559        StreamEntry[] entries,
 560        RedisValue consumerName,
 561        BatchProgress progress,
 562        CancellationToken stoppingToken,
 563        CancellationToken cancellationToken)
 564    {
 565        // ~PendingMessageMinIdleTime/3: two chances to land an idle reset inside every reclaim
 566        // window even when one sweep is delayed by a slow round trip.
 567        var interval = TimeSpan.FromMilliseconds(Math.Max(1, SubscriberOptions.PendingMessageMinIdleTime.TotalMillisecon
 568        try
 569        {
 570            while (true)
 571            {
 572                await Task.Delay(interval, cancellationToken).ConfigureAwait(false);
 573
 574                // Claim from the first unsettled entry onward: that covers the entry currently in
 575                // the handler plus everything still waiting its turn. minIdle 0 resets the idle
 576                // clock unconditionally, and the races are harmless — an entry ACKed while this
 577                // sweep is in flight has left the PEL (the claim simply skips it), and a failed
 578                // entry is settled before MarkSettled runs, so its post-failure idle countdown is
 579                // never stretched. Once stopping — the token, or the worker's intake gate — only the
 580                // entry in the handler is kept: the rest will not start here, so its idle clock is
 581                // left to run for a peer's reclaim.
 582                var settled = progress.SettledCount;
 583                if (settled >= entries.Length)
 584                    return;
 585
 586                var end = stoppingToken.IsCancellationRequested || IntakeClosed ? settled + 1 : entries.Length;
 587                var remaining = new RedisValue[end - settled];
 588                for (var i = settled; i < end; i++)
 589                    remaining[i - settled] = entries[i].Id;
 590
 591                try
 592                {
 593                    await _database.StreamClaimIdsOnlyAsync(
 594                        Stream,
 595                        ConsumerGroup,
 596                        consumerName,
 597                        minIdleTimeInMilliseconds: 0,
 598                        remaining,
 599                        cancellationToken).ConfigureAwait(false);
 600                }
 601                catch (Exception ex) when (ex is not OperationCanceledException)
 602                {
 603                    Logger.LogWarning(
 604                        ex,
 605                        "Failed to refresh the pending idle time of {EntryCount} Redis entries on {Stream}; a sibling ma
 606                        remaining.Length,
 607                        Stream.ToString());
 608                }
 609            }
 610        }
 611        catch (OperationCanceledException)
 612        {
 613            // The dispatch loop let go of the batch.
 614        }
 615    }

 617    private sealed class BatchProgress
 618    {
 619        // Settled only ever increments, so a monotonic volatile read is enough — no lock, and a
 620        // stale read only claims an already-settled entry once more (an ACKed id is simply
 621        // absent from the PEL, and a failed one gets its reclaim delayed by one sweep).
 622        private int _settledCount;
 623
 624        public int SettledCount => Volatile.Read(ref _settledCount);
 625
 626        public void MarkSettled() => Interlocked.Increment(ref _settledCount);
 627    }

 675    /// <summary>Length budget of the generated consumer name, before the role suffix.</summary>

 678    /// <summary>

 689    {
 690        var suffix = $"-{processId.ToString(System.Globalization.CultureInfo.InvariantCulture)}-{instance:N}";
 691        return PortableText.TruncateWellFormed(machineName, MaxGeneratedConsumerNameLength - suffix.Length) + suffix;
 692    }
