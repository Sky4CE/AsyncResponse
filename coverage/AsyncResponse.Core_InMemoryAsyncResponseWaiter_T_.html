 440    private static Task DispatchManyAsync<TState>(
 441        SubscriptionBase[]? subscribers,
 442        Func<SubscriptionBase, TState, Task> dispatch,
 443        TState state)
 444    {
 445        if (subscribers is null || subscribers.Length == 0)
 446            return Task.CompletedTask;
 447
 448        Task? firstPending = null;
 449        List<Task>? pending = null;
 450        for (var i = 0; i < subscribers.Length; i++)
 451        {
 452            var task = dispatch(subscribers[i], state);
 453            if (task.IsCompletedSuccessfully)
 454                continue;
 455
 456            if (firstPending is null)
 457            {
 458                firstPending = task;
 459                continue;
 460            }
 461
 462            (pending ??= [firstPending]).Add(task);
 463        }
 464
 465        return pending is not null
 466            ? Task.WhenAll(pending)
 467            : firstPending ?? Task.CompletedTask;
 468    }

 494    {
 495        private readonly object _gate = new();
 496        private SubscriptionBase? _single;
 497        private List<SubscriptionBase>? _many;
 498        private bool _closed;
 499
 500        public int Count
 501        {
 502            get
 503            {
 504                lock (_gate)
 505                    return _single is not null ? 1 : _many?.Count ?? 0;
 506            }
 507        }

 509        /// <summary>Adds a subscription to this correlation-id group.</summary>
 510        public bool TryAdd(SubscriptionBase subscription)
 511        {
 512            lock (_gate)
 513            {
 514                if (_closed)
 515                    return false;
 516
 517                if (_single is null && _many is null)
 518                {
 519                    _single = subscription;
 520                    return true;
 521                }
 522
 523                if (_many is null)
 524                {
 525                    _many = [_single!, subscription];
 526                    _single = null;
 527                    return true;
 528                }
 529
 530                _many.Add(subscription);
 531                return true;
 532            }
 533        }
 534
 535        /// <summary>Removes a subscription and returns whether the group became empty.</summary>
 536        public bool Remove(Guid subscriptionId)
 537        {
 538            lock (_gate)
 539            {
 540                if (_single?.Id == subscriptionId)
 541                {
 542                    _single = null;
 543                    _closed = true;
 544                    return true;
 545                }
 546
 547                if (_many is null)
 548                    return false;
 549
 550                for (var i = 0; i < _many.Count; i++)
 551                {
 552                    if (_many[i].Id != subscriptionId)
 553                        continue;
 554
 555                    _many.RemoveAt(i);
 556                    // _many is only ever created with two entries and collapses to _single at one,
 557                    // so it can never reach zero here — the group-empty signal is produced solely
 558                    // by the _single removal path above.
 559                    if (_many.Count == 1)
 560                    {
 561                        _single = _many[0];
 562                        _many = null;
 563                    }
 564
 565                    return false;
 566                }
 567
 568                return false;
 569            }
 570        }

 613    private abstract class SubscriptionBase
 614    {

 616        private readonly CancellationTokenSource _timeoutCts;

 618        private readonly object _cleanupSync = new();

 620        private CancellationTokenRegistration _timeoutRegistration;
 621        private Task? _cleanupTask;

 627        /// <summary>Creates the common state for an in-memory waiter subscription.</summary>
 628        protected SubscriptionBase(InMemoryAsyncResponseChannel owner, string correlationId, TimeSpan timeout, Activity?
 629        {

 632            Timeout = timeout;
 633            _activity = activity;

 635        }
 636
 637        /// <summary>Per-waiter registration id used for subscription and recovery-state cleanup.</summary>
 638        public Guid Id { get; } = Guid.NewGuid();
 639        protected string CorrelationId { get; }
 640        protected Activity? WaitActivity => _activity;
 641        private TimeSpan Timeout { get; }
 642        public bool CleanupStarted
 643        {

 645            get => Volatile.Read(ref _cleanupStarted) != 0;

 648        /// <summary>Arms the subscription timeout after registration has succeeded.</summary>
 649        public void ArmTimeout()
 650        {
 651            if (CleanupStarted)
 652                return;
 653
 654            try
 655            {
 656                _timeoutRegistration = _timeoutCts.Token.Register(static state =>
 657                {
 658                    _ = ((SubscriptionBase)state!).TimeoutAsync();
 659                }, this);
 660
 661                if (CleanupStarted)
 662                {
 663                    _timeoutRegistration.Dispose();
 664                    return;
 665                }
 666
 667                _timeoutCts.CancelAfter(Timeout);
 668            }
 669            catch (ObjectDisposedException)
 670            {
 671                // A response or explicit disposal can clean up between the guard and timer arming.
 672            }
 673        }

 676        /// Dispatches a typed or materializable response to this subscription.

 679        /// waiter materializes its own instance; <c>null</c> when the response is a

 690                exception,

 693        private Task DispatchExceptionCoreAsync(Exception exception)
 694        {
 695            if (CleanupStarted)
 696                return Task.CompletedTask;
 697
 698            if (!TryBeginTerminal())
 699                return Task.CompletedTask;
 700

 702            TrySetException(exception);

 706        /// <summary>
 707        /// Serializes every signal for one waiter. The uncontended path uses only an interlocked
 708        /// owner bit; the semaphore is created lazily if concurrent publishers actually contend.
 709        /// </summary>
 710        protected Task DispatchSerialAsync<TState>(
 711            TState state,
 712            Func<SubscriptionBase, TState, Task> dispatch)
 713        {
 714            if (Interlocked.CompareExchange(ref _dispatching, 1, 0) != 0)
 715                return WaitAndDispatchAsync(this, state, dispatch);
 716
 717            Task task;
 718            try
 719            {
 720                task = dispatch(this, state);
 721            }
 722            catch
 723            {
 724                ReleaseDispatch();
 725                throw;
 726            }
 727
 728            if (task.IsCompletedSuccessfully)
 729            {
 730                ReleaseDispatch();
 731                return task;
 732            }
 733
 734            return ReleaseAfterDispatchAsync(this, task);
 735        }

 737        private static async Task WaitAndDispatchAsync<TState>(
 738            SubscriptionBase subscription,
 739            TState state,
 740            Func<SubscriptionBase, TState, Task> dispatch)
 741        {
 742            Interlocked.Increment(ref subscription._dispatchWaiterCount);
 743            try
 744            {
 745                var waiters = LazyInitializer.EnsureInitialized(
 746                    ref subscription._dispatchWaiters,
 747                    static () => new SemaphoreSlim(0));
 748                while (Interlocked.CompareExchange(ref subscription._dispatching, 1, 0) != 0)
 749                    await waiters.WaitAsync().ConfigureAwait(false);
 750            }
 751            finally
 752            {
 753                Interlocked.Decrement(ref subscription._dispatchWaiterCount);
 754            }
 755
 756            try
 757            {
 758                await dispatch(subscription, state).ConfigureAwait(false);
 759            }
 760            finally
 761            {
 762                subscription.ReleaseDispatch();
 763            }
 764        }

 767        {
 768            try
 769            {
 770                await task.ConfigureAwait(false);
 771            }
 772            finally
 773            {
 774                subscription.ReleaseDispatch();
 775            }
 776        }

 779        private void ReleaseDispatch()

 786        /// <summary>Runs subscription, recovery-state, timeout, and activity cleanup once.</summary>
 787        public ValueTask CleanupOnceAsync()
 788        {
 789            Task cleanupTask;
 790            lock (_cleanupSync)
 791            {
 792                cleanupTask = _cleanupTask ??= StartCleanupAsync();
 793            }
 794
 795            return cleanupTask.IsCompletedSuccessfully
 796                ? ValueTask.CompletedTask
 797                : new ValueTask(cleanupTask);
 798        }

 800        /// <summary>
 801        /// Dispose-path cleanup: DRAINS any in-flight dispatch before settling. A delivery may be
 802        /// mid <c>Until</c>-predicate holding a claimed terminal message; queueing a no-op through
 803        /// the per-waiter dispatch gate completes only after that delivery settled the task (or
 804        /// released the gate), so the cleanup's cancel afterwards is a genuine settlement — never
 805        /// a cancellation stealing an already-consumed response. Must NOT be called from dispatch
 806        /// code (which holds the gate): dispatch-triggered cleanup uses
 807        /// <see cref="CleanupOnceAsync"/> directly, with its task already settled.
 808        /// <para>
 809        /// The drain is bounded by <c>DisposalDrainTimeout</c>. A lapsed budget must not fall back
 810        /// to the cleanup's cancel — the wedged delivery holds a message the channel already
 811        /// claimed, and "canceled" would tell a re-attaching caller nothing was delivered — so it
 812        /// faults the task with the explicit indeterminate contract instead, routing durable flows
 813        /// to a fresh idempotent restart. The abandoned no-op marker runs harmlessly whenever the
 814        /// wedged dispatch finally releases the gate.
 815        /// </para>
 816        /// </summary>
 817        public async ValueTask DisposeCleanupAsync()
 818        {
 819            if (Volatile.Read(ref _cleanupStarted) == 0)
 820            {
 821                var drainTimeout = _owner._options.DisposalDrainTimeout;
 822                try
 823                {
 824                    await DispatchSerialAsync(0, static (_, _) => Task.CompletedTask)
 825                        .WaitAsync(drainTimeout).ConfigureAwait(false);
 826                }
 827                catch (TimeoutException)
 828                {
 829                    _owner._logger.LogWarning(
 830                        "Disposal drain for correlationId {CorrelationId} did not finish within {DrainTimeout}; faulting
 831                        CorrelationId, drainTimeout);
 832                    AsyncResponseDiagnostics.SetError(_activity, "indeterminate_delivery", "Disposal drain timed out wit
 833                    // A TrySetResult from the late-finishing dispatch loses against this and is
 834                    // dropped; its cleanup call is a no-op behind the latch.
 835                    TrySetException(new AsyncResponseIndeterminateDeliveryException(CorrelationId, drainTimeout));
 836                }
 837            }
 838
 839            await CleanupOnceAsync().ConfigureAwait(false);
 840        }

 842        private async Task StartCleanupAsync()
 843        {
 844            Volatile.Write(ref _cleanupStarted, 1);
 845
 846            // A waiter disposed before any terminal signal must not leave ResponseTask pending
 847            // forever for callers that hold it directly. Cancellation is a no-op after a normal
 848            // completion, timeout, or fault.
 849            TrySetCanceled();
 850
 851            try
 852            {
 853                // Delete the recovery state BEFORE removing the subscription. In the reverse order
 854                // a publish landing in the window sees "no subscriber, state present" and fires a
 855                // spurious recovery callback for a wait that already reached a terminal state. In
 856                // this order the window shows a subscriber that drops the message (CleanupStarted)
 857                // — a late or duplicate terminal message is droppable; a resurrected recovery
 858                // callback is not.
 859                await _owner._recoveryStateStore.TryDeleteAsync(CorrelationId, Id).ConfigureAwait(false);
 860            }
 861            finally
 862            {
 863                _owner.RemoveSubscription(CorrelationId, Id);
 864                await _timeoutRegistration.DisposeAsync().ConfigureAwait(false);
 865                _timeoutCts.Dispose();
 866                _activity?.Dispose();
 867            }
 868        }
