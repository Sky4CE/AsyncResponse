 317            transport.Readmit(queued);
 318
 319        var now = Clock.GetUtcNow().UtcDateTime;
 320        foreach (var job in pendingDelayed)
 321        {
 322            var remaining = job.NotBeforeUtc is { } notBefore && WorkerJobExecutor.AsUtc(notBefore) > now
 323                ? WorkerJobExecutor.AsUtc(notBefore) - now
 324                : TimeSpan.Zero;
 325            if (remaining > TimeSpan.Zero)
 326            {
 327                // Per-hop clamp, as every production publisher applies: NotBeforeUtc rides the
 328                // envelope, so the executor re-delays the remainder on delivery. Unclamped, a
 329                // legal 60-day sleep would throw here and silently lose the rest of the list.
 330                // Scheduled without waiting for a delayed slot: the drain frees every slot
 331                // before it retains, so the retained set can exceed DelayedJobCapacity, and a
 332                // slot only frees when a timer fires — on a clock that cannot move while the
 333                // test is in here.
 334                var hop = remaining <= transport.MaxPublishDelay ? remaining : transport.MaxPublishDelay;
 335                transport.ScheduleRetained(job, hop);
 336            }
 337            else
 338                transport.Readmit(job);
 339        }
 340    }

 344    private void BuildProvider()
 345    {
 346        var services = new ServiceCollection();
 347
 348        // The engine clock, the shared durable state, and the harness observers go in FIRST so the
 349        // TryAdd registrations inside AddAsyncResponse()/With*() adopt them.
 350        services.AddSingleton<TimeProvider>(Clock);
 351        services.AddSingleton(_recoveryStore);
 352        services.AddSingleton(_flowStore);
 353        services.AddSingleton<IDurableFlowExecutionObserver>(_quiesce);
 354        foreach (var observer in _observers)
 355            services.AddSingleton<IDurableFlowExecutionObserver>(observer);
 356
 357        _options.ConfigureServices?.Invoke(services);
 358
 359        var builder = services.AddAsyncResponse()
 360            .WithInMemoryChannel(channel =>
 361            {
 362                _options.Channel?.Invoke(channel);
 363            })
 364            .WithInMemoryTransport(transport =>
 365            {
 366                _options.Transport?.Invoke(transport);
 367            })
 368            .WithInMemoryDurableFlows(flows =>
 369            {
 370                _options.DurableFlows?.Invoke(flows);
 371            });
 372
 373        _options.ConfigureAsyncResponse?.Invoke(builder);
 374
 375        // Both checks run after EVERY user hook — ConfigureAsyncResponse included, whose builder
 376        // exposes the same service collection: run before it, a clock or a logging registration
 377        // made there slipped past them.
 378        //
 379        // The engine resolves the LAST TimeProvider registration, so a clock registered by the
 380        // test would silently displace the virtual one: no engine timer ever arms, AdvanceAsync
 381        // advances a clock nothing reads, and every wait dies as an unexplained RealTimeGuard
 382        // timeout. Fail construction instead, naming the fix.
 383        var lastClock = services.LastOrDefault(d => !d.IsKeyedService && d.ServiceType == typeof(TimeProvider));
 384        if (lastClock is null || !ReferenceEquals(lastClock.ImplementationInstance, Clock))
 385        {
 386            throw new InvalidOperationException(
 387                $"{nameof(AsyncResponseTestHarness)} drives the whole engine on its own virtual clock; a TimeProvider " 
 388                "registered via ConfigureServices or ConfigureAsyncResponse would displace it and no timer, timeout, lea
 389                "or backoff would ever elapse. Use harness.Clock / AdvanceAsync instead of registering your own TimeProv
 390        }
 391
 392        // Fallback only, and only AFTER the user's registrations: AddLogging registers ILogger<>
 393        // with TryAdd semantics, so a non-Try registration made before them would silently pin
 394        // NullLogger and swallow the very diagnostics the harness's failure messages tell users
 395        // to check.
 396        services.TryAddSingleton(typeof(ILogger<>), typeof(NullLogger<>));
 397
 398        _provider = services.BuildServiceProvider(new ServiceProviderOptions { ValidateScopes = true });
 399
 400        // The probe needs the engine clock and this incarnation's timer settings to tell an
 401        // in-process timer park (holds a worker slot) from a suspension (the job ends), and to
 402        // know when a park's hop ends — see QuiesceProbe.OnStepWaitingAsync.
 403        var flowOptions = _provider.GetRequiredService<DurableFlowOptions>();
 404        _quiesce.Arm(Clock, flowOptions.TimerInProcessThreshold, flowOptions.MaxInProcessParkDuration);
 405
 406        // The flow probe makes the same in-process/suspended distinction: only an in-process
 407        // park dies with its incarnation, so only its Waiting event is hidden from a barrier
 408        // after a restart.
 409        foreach (var observer in _observers)
 410            (observer as FlowProbe)?.Arm(Clock, flowOptions.TimerInProcessThreshold);
 411    }

 421    private async Task StopHostedServicesAsync()
 422    {
 423        // Bounded by the real-time guard: a graceful stop can legitimately hang when a drained job
 424        // is parked on virtual time; the restart then proceeds like a hard crash and the lease
 425        // machinery reconciles the leftover execution.
 426        using var cutoff = new CancellationTokenSource(_options.RealTimeGuard);
 427
 428        // An engine-owned park — an awaited step, an in-process timer on the virtual clock — ends
 429        // only when the test replies or moves the clock, and neither can happen while the test is
 430        // awaiting this stop. Waiting it out therefore always burned the WHOLE guard (10 s by
 431        // default), once per restart and again per disposal, in a kit whose point is that nothing
 432        // sleeps for real. The moment a park is all that is left, end the wait: that is where the
 433        // lapsed guard was heading anyway — the leases are broken right after and the new
 434        // incarnation takes the execution over, exactly as after a real crash.
 435        using var stopWatching = new CancellationTokenSource();
 436        var watcher = AbandonOnceOnlyParkedAsync(cutoff, stopWatching.Token);
 437        try
 438        {
 439            foreach (var service in Enumerable.Reverse(_started))
 440            {
 441                try
 442                {
 443                    await service.StopAsync(cutoff.Token).ConfigureAwait(false);
 444                }
 445                catch (OperationCanceledException)
 446                {
 447                    // Hard-crash semantics for whatever did not stop in time.
 448                }
 449            }
 450        }
 451        finally
 452        {
 453            await stopWatching.CancelAsync().ConfigureAwait(false);
 454            await watcher.ConfigureAwait(false);
 455        }
 456
 457        _started = [];
 458    }

 460    /// <summary>
 461    /// Cancels <paramref name="cutoff"/> as soon as every execution still running is waiting on
 462    /// the virtual clock or a reply — an engine-owned park, or a crashed attempt asleep in a
 463    /// redelivery backoff the drain took — and no queued job can still get a worker. Deliberately
 464    /// does nothing while NOTHING is waiting that way: a stop that can still make progress is left
 465    /// to finish cleanly, and the real-time guard stays the backstop for user code that is
 466    /// genuinely stuck. A backoff taken BEFORE the stop is no such wait: it is bound to the worker
 467    /// host's stopping token, so the host's own stop ends it (dropping its job) and drains what is
 468    /// queued behind it — this check runs before any hosted service has stopped, and counting it
 469    /// cut that stop short at once.
 470    /// </summary>
 471    private async Task AbandonOnceOnlyParkedAsync(CancellationTokenSource cutoff, CancellationToken stopWatching)
 472    {
 473        var transport = Transport;
 474        try
 475        {
 476            while (!stopWatching.IsCancellationRequested)
 477            {
 478                // Queued jobs never started and run no user code, but while a worker is free they
 479                // still make progress on their own; once every worker is held they wait on the
 480                // same clock (or reply) as whatever holds it.
 481                var executing = transport.ExecutingJobs;
 482                var queuedCanRun = transport.OutstandingJobs > executing && executing < transport.Options.WorkerCount;
 483                var waiting = _quiesce.ParkedCount + transport.DrainBackingOffJobs;
 484                if (waiting > 0 && !queuedCanRun && UserCodeRunning(transport) <= 0)
 485                {
 486                    await cutoff.CancelAsync().ConfigureAwait(false);
 487                    return;
 488                }
 489
 490                // The SYSTEM clock: the harness's virtual one is not moving while a test awaits
 491                // this stop, which is the whole reason the park cannot end by itself.
 492                await Task.Delay(ParkedStopPollInterval, TimeProvider.System, stopWatching).ConfigureAwait(false);
 493            }
 494        }
 495        catch (OperationCanceledException)
 496        {
 497            // The stop finished on its own.
 498        }
 499    }

 501    /// <summary>How often the stop checks whether a park is all that is left.</summary>
 502    private static readonly TimeSpan ParkedStopPollInterval = TimeSpan.FromMilliseconds(5);
 503
 504    /// <summary>
 505    /// Executions of <paramref name="transport"/>'s incarnation still running user code: jobs a
 506    /// worker has taken (queued ones never started) plus inline direct runs, minus the ones
 507    /// waiting on the virtual clock or a reply — engine-owned parks and crashed attempts asleep in
 508    /// a redelivery backoff the drain took. A backoff taken before the stop counts as running: the
 509    /// worker host's stop ends it and the job then leaves (see <see cref="AbandonOnceOnlyParkedAsync"/>).
 510    /// </summary>
 511    private int UserCodeRunning(InMemoryWorkerTransport transport)
 512        => transport.ExecutingJobs - transport.DrainBackingOffJobs + _quiesce.DirectRunsInFlight - _quiesce.ParkedCount;

 525    /// <summary>
 526    /// Give worker jobs that are still RUNNING code a real chance to reach their next stable
 527    /// state before the clock moves. Advancing while a job is mid-code would let a later
 528    /// <c>DelayAsync</c> or deadline be computed from the already-advanced clock and park beyond
 529    /// the advance target — the load-dependent flake this settle exists to prevent. Three
 530    /// signals, cheapest first:
 531    /// <list type="bullet">
 532    /// <item>Every outstanding job is parked on an event-visible engine wait (an awaited reply,
 533    /// an in-process flow timer) — the quiesce probe's count covers the outstanding count, and
 534    /// settling costs nothing.</item>
 535    /// <item>A virtual timer was armed while settling — the busy job just began a virtual-time
 536    /// wait (a retry backoff, a lease-acquisition poll, a timer chunk), so advancing the clock is
 537    /// exactly how it progresses.</item>
 538    /// <item>A bounded real-time grace for what cannot be attributed (a job blocked on a virtual
 539    /// timer it armed before this settle began looks identical to one stuck in user code). The
 540    /// budget is generous relative to scheduling noise; a fake that sleeps on the SYSTEM clock
 541    /// longer than this is the documented harness anti-pattern (use the injected TimeProvider).</item>
 542    /// </list>
 543    /// </summary>
 544    /// <summary>
 545    /// Reports an inline executor attempt starting (see QuiesceProbe.DirectRunsInFlight); pass the
 546    /// returned token to <see cref="OnDirectRunFinished"/>.
 547    /// </summary>
 548    internal int OnDirectRunStarted() => _quiesce.OnDirectRunStarted();
 549
 550    /// <summary>Reports an inline executor attempt finished.</summary>
 551    internal void OnDirectRunFinished(int token) => _quiesce.OnDirectRunFinished(token);

 553    private async Task SettleAsync()
 554    {
 555        // Let just-released continuations (a fired timer's write, worker dispatch) reach the
 556        // transport counters before reading them.
 557        for (var round = 0; round < 3; round++)
 558        {
 559            await Task.Yield();
 560            await Task.Delay(TimeSpan.FromMilliseconds(1)).ConfigureAwait(false);
 561        }
 562
 563        var budget = TimeProvider.System.GetUtcNow() + TimeSpan.FromMilliseconds(500);
 564        while (Transport.OutstandingJobs + _quiesce.DirectRunsInFlight > _quiesce.ParkedCount
 565               && TimeProvider.System.GetUtcNow() < budget)
 566        {
 567            // A new earliest due time counts only when a timer was actually ARMED: disposing the
 568            // earliest timer (a finished wait, a cancelled delay) moves NextTimerDueAt too, and
 569            // ended the settle while the job that disposed it was still running code — the clock
 570            // then advanced under it. (Any arm alone is not enough either: a job that just took a
 571            // worker arms its lease-renew timer long before it reaches a wait of its own.)
 572            var armed = Clock.ArmSequence;
 573            var next = Clock.NextTimerDueAt;
 574            await Task.Delay(TimeSpan.FromMilliseconds(1)).ConfigureAwait(false);
 575            if (armed != Clock.ArmSequence && next != Clock.NextTimerDueAt)
 576                return; // A virtual wait just began — the advance loop re-evaluates immediately.
 577        }
 578    }

 580    /// <summary>
 581    /// Tracks flow executions parked on an engine-owned wait: between a step's Waiting event and
 582    /// its Completed event the execution holds a worker slot but is idle by design, waiting on
 583    /// virtual time or a test-published reply. Only waits that actually park in process count —
 584    /// child-flow steps always suspend (the job ends), and a timer whose remainder exceeds the
 585    /// in-process threshold suspends too on this delayed-capable transport, so its Waiting is not
 586    /// counted (a stale entry would outlive the worker slot until the wake-up replay and mask a
 587    /// genuinely busy job in SettleAsync's guard). Cleared on restart: the old incarnation's
 588    /// parked executions die with it.
 589    /// </summary>
 590    private sealed class QuiesceProbe : IDurableFlowExecutionObserver
 591    {
 592        /// <summary>
 593        /// Parked steps, each with the instant its in-process wait ENDS: <c>null</c> for an awaited
 594        /// step (it ends on a reply), the hop's end for a timer.
 595        /// </summary>
 596        private readonly Dictionary<(string FlowId, string Step), DateTime?> _parked = [];
 597        private readonly object _directGate = new();
 598        private TimeProvider _clock = TimeProvider.System;
 599        private TimeSpan _timerInProcessThreshold;
 600        private TimeSpan? _maxInProcessParkDuration;
 601        private int _directRunsInFlight;
 602        private int _directGeneration;
 603
 604        /// <summary>
 605        /// Steps parked right now. A timer entry stops counting once the clock reaches its hop's
 606        /// end: the wait has fired and the job is running code again — completing the step, or
 607        /// handing the timer over to a fresh delivery, which ends the job without any observer
 608        /// event (a suspension notifies nothing), so the entry cannot be removed there.
 609        /// </summary>
 610        public int ParkedCount
 611        {
 612            get
 613            {
 614                var now = _clock.GetUtcNow().UtcDateTime;
 615                lock (_parked)
 616                    return _parked.Values.Count(waitEnd => waitEnd is not { } end || end > now);
 617            }
 618        }
 619
 620        /// <summary>
 621        /// Executor attempts running inline (<c>FlowRunHandle.ExecuteDirectAsync</c>), which hold
 622        /// no worker slot: SettleAsync counts them beside <c>Transport.OutstandingJobs</c>, since
 623        /// a step they park would otherwise make ParkedCount exceed the outstanding jobs and let
 624        /// the clock advance under a direct run still executing user code.
 625        /// </summary>
 626        public int DirectRunsInFlight
 627        {
 628            get { lock (_directGate) return _directRunsInFlight; }
 629        }
 630
 631        /// <summary>Counts a direct run in; returns the generation its finish must match.</summary>
 632        public int OnDirectRunStarted()
 633        {
 634            lock (_directGate)
 635            {
 636                _directRunsInFlight++;
 637                return _directGeneration;
 638            }
 639        }
 640
 641        /// <summary>
 642        /// Counts a direct run out — unless a restart reset the count since it started: a direct
 643        /// run the restart abandoned (a parked one, which does not stop a restart) ends only after
 644        /// the reset, and decrementing then drove the count to -1, so every later check
 645        /// under-counted one busy direct run until the next restart.
 646        /// </summary>
 647        public void OnDirectRunFinished(int generation)
 648        {
 649            lock (_directGate)
 650            {
 651                if (generation == _directGeneration)
 652                    _directRunsInFlight--;
 653            }
 654        }

 656        /// <summary>Binds the engine clock and the in-process timer settings of the current incarnation.</summary>
 657        public void Arm(TimeProvider clock, TimeSpan timerInProcessThreshold, TimeSpan? maxInProcessParkDuration)
 658        {
 659            _clock = clock;
 660            _timerInProcessThreshold = timerInProcessThreshold;
 661            _maxInProcessParkDuration = maxInProcessParkDuration;
 662        }
 663
 664        public void Reset()
 665        {
 666            lock (_parked) _parked.Clear();
 667            lock (_directGate)
 668            {
 669                _directGeneration++;
 670                _directRunsInFlight = 0;
 671            }
 672        }

 674        public ValueTask OnStepWaitingAsync(DurableFlowStepEvent step)
 675        {
 676            // Awaited steps always park in process holding their worker slot. Timer steps only
