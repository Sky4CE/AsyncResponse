 627        // Counted BEFORE the stop check (DisposeAsync sets the flag, then reads the count), so a

 631        {
 632            // A stopping subscriber starts nothing new: RabbitMQ.Client keeps dispatching the

 634            // now would be cut off by the channel close anyway — its side effects run, the broker
 635            // requeues it, and a peer runs it again. Left un-ACKed, it is redelivered unstarted.
 636            if (subscriberCancellationToken.IsCancellationRequested || Volatile.Read(ref _stopping) != 0)
 637                return;

 639            // Host stop has begun (a worker subscriber only): the flow engine now hands back any
 640            // timer wait that starts here, so a delivery started now would spend an attempt on a

 652    }

 658    /// the job a second time on every deploy. Past the bound the close goes ahead, and the broker

 705        // FailFast, a hang that tripped the broker's consumer_timeout — is requeued by the broker

 747        }
